Anti-cheating Engine for Android

info
  • This document outlines an Android anti-cheating engine analyzed and developed in 2016.
    • https://github.com/Lazenca/Lazenca-S
  • The techniques described here represent fundamental, basic defensive mechanisms.
  • Modern, advanced cheat engines and dynamic binary instrumentation frameworks may bypass these checks.

Check Modification of Binaries (Integrity Verification)​

  • Verifying binary integrity helps detect tampered application binaries and shared libraries:
    • Compute cryptographic hashes (e.g., SHA-256) of target local files and compare them against authoritative values sent from the server.
  • Important implementation considerations:
    • Target file paths must be obfuscated or dynamically provided from the server.
      • If file paths are hardcoded as plain strings in code or assets, reverse engineers can easily modify paths or bypass the hash comparison logic.
    • Do not use weak or collision-prone hashing algorithms like CRC32/MD5 for tamper detection.
CheckBinaryModification.h
bool ModFindHash(char *hash) {

	long lSize;
	size_t result;
	int result_val;

	char *buffer;
	char patternFile[MAX_PATH] = "/mnt/sdcard/pattern";

	FILE *fp = fopen(patternFile, "rb");

	if (fp) {
		fseek(fp, 0, SEEK_END);
		lSize = ftell(fp);
		rewind(fp);

		buffer = (char*) malloc(sizeof(char) * lSize);
		if (buffer == NULL) {
			fputs("Memory error", stderr);
			exit(2);
		}

		result = fread(buffer, 1, lSize, fp);
		if (result != lSize) {
			fputs("Reading error", stderr);
			exit(3);
		}

		if (result > 0) {
			if(strstr(buffer, hash)){
				result_val = true;
			}else{
				result_val = false;
			}
		}
		fclose(fp);
		free(buffer);
	} else {
		return false;
	}
	return result_val;
}

void CheckModBinary(){
	pid_t pid = getpid();

	char sha256[65] = "";

	char packageName[MAX_PATH];
	char apkFilePath[MAX_PATH];
	char libFilePath[MAX_PATH];
	char tmpLibFilePath[MAX_PATH];

	int zipcount = 0;
	unsigned int crc;

	DbgPrint("Modification Check");

	getCmdline(pid, packageName, sizeof(packageName));
	getApkFilePath(packageName, &apkFilePath);

	sprintf(libFilePath, "/data/data/%s/lib", packageName);

	while(1)
	{
		fileToSha256(&sha256, apkFilePath);

		if(ModFindHash(sha256) == false)
		{
			DbgPrint("File Path : %s, SHA256 : %s - Modification", apkFilePath, sha256);
			DbgPrint("Modification.");
			sleep(10);
			exit(0);
		}

		DIR *dirInfo;
		struct dirent *dirEntry;

		dirInfo = opendir(libFilePath);
		if (NULL != dirInfo) {
			while (dirEntry = readdir(dirInfo))
			{
				sprintf(tmpLibFilePath, "/data/data/%s/lib/%s", packageName, dirEntry->d_name);

				fileToSha256(&sha256, tmpLibFilePath);
				if (ModFindHash(sha256) == false) {
					DbgPrint(".SO File Hash Check - Modification.");
					sleep(10);
					exit(0);
				}
			}
			closedir(dirInfo);
		}
		sleep(1000);
	}
}

Check Debugging (Anti-Debugging)​

  • Debugging status can be detected using two primary techniques:
    • 1. Checking the Parent Process Name via /proc/pid/cmdline:
      • If the application is launched by GDB, its parent process (PPID) is gdb. Reading /proc/<ppid>/cmdline allows detecting whether GDB spawned the process.
bool isCheckCmdline()
bool isCheckCmdline() {
   char filePath[32], fileRead[128];
   FILE* file;

   snprintf(filePath, 24, "/proc/%d/cmdline", getppid());
   file = fopen(filePath, "r");

   fgets(fileRead, 128, file);
   fclose(file);

   if(!strcmp(fileRead, "gdb")) {
	   DbgPrint("Debugger(gdb) detected\n");
       return true;
   }
   DbgPrint("Clear(Debug)\n");
   return false;
}
  • 2. Inspecting the TracerPid Field in /proc/pid/status:
    • The /proc/self/status file contains process metrics including thread counts, memory limits, and signal masks.
    • The TracerPid field indicates the PID of a tracing process (such as a debugger attached via ptrace).
      • If TracerPid != 0, a tracer/debugger is actively attached.
      • If TracerPid == 0, the process is currently untraced.
bool isCheckTracerPid()
bool isCheckTracerPid() {
	int TPid;
	char buf[512];

	const char *str = "TracerPid:";
	size_t strSize = strlen(str);

	FILE* file = fopen("/proc/self/status", "r");

	while (fgets(buf, 512, file)) {
		if (!strncmp(buf, str, strSize)) {
			sscanf(buf, "TracerPid: %d", &TPid);
			if (TPid != 0) {
				DbgPrint("Debugger detected\n");
				return true;
			}
		}
	}
	fclose(file);
	DbgPrint("Clear(Debug)\n");
	return false;
}

Check Speed Hacking​

  • Speed hack engines in cheat tools hook system time functions (gettimeofday, clock_gettime, etc.) to manipulate game speed.
  • Speed hacks can be detected by measuring elapsed system time across known sleep durations:
    • Capture a baseline start timestamp from system time APIs.
    • Call sleep() (or nanosleep()) to pause the process for a fixed interval.
    • Capture an end timestamp immediately after waking.
    • Calculate elapsed time (end - start) and compare against the expected sleep interval.
    • If the difference between measured time and the sleep duration exceeds acceptable variance thresholds, speed tampering is detected.
    • Example:
      • For gettimeofday() after sleep(100), the elapsed duration should fall within expected lower/upper bounds.
      • For clock_gettime(CLOCK_MONOTONIC) after sleep(100), measured nanoseconds should closely match expected sleep ticks.
  • Note: Modern cheat engines frequently hook all time and sleep functions in tandem, which may evade basic tick comparisons.
CheckSpeedHack.h
int getTime1()
{
	struct timeval tv;
	struct timezone tz;
	gettimeofday (&tv, &tz);
	return (tv.tv_sec*1000 + tv.tv_usec/1000);
}

long getTime2()
{
	struct timespec now;
	clock_gettime(CLOCK_MONOTONIC, &now);
	return now.tv_sec*1000000 + now.tv_nsec/1000;
}

void CheckSpeedHack(){
	int start = 0, end = 0, time_data = 0;
	int start2 = 0, end2 = 0, time_data2 = 0;
	int cmp = 1;

	while(cmp){
		start = getTime1();
		start2 = getTime2();

		sleep(100);
		end = getTime1();
		end2 = getTime2();

		time_data = end - start;
		time_data2 = end2 - start2;
		DbgPrint("time_data : %d, end : %d , start : %d", time_data, end, start);
		DbgPrint("time_data2 : %d, end2 : %d , start2 : %d", time_data2, end2, start2);
		if(time_data > 101000 || time_data < 10000){
			cmp = 0;
			DbgPrint("SpeedHackDetect");
			sleep(10);
			exit(0);
		}
		if(time_data2 > 100001000 || time_data2 < 100000000){
			cmp = 0;
			DbgPrint("SpeedHackDetect");
			sleep(10);
			exit(0);
		}
	}
}

Check Virtual Machine (Emulator Detection)​

  • Detecting Android emulators / virtual machines:
    • Inspect system properties in /system/build.prop.
    • The /system/build.prop file stores configuration and hardware properties of the Android system.
    • Emulators typically populate distinctive strings in properties such as:
      • ro.product.manufacturer
      • ro.product.model
      • ro.product.name
      • ro.product.device
    • Note: Rooted or custom emulator images can spoof these property values to evade simple string checks.
  • Additional emulator detection techniques:
    • Checking for known emulator-specific package names, services, or driver devices (e.g., /dev/qemu_pipe, /dev/socket/qemud).
    • Inspecting CPU features and QEMU-specific hardware markers.
CheckVirtualMachine.h
void virtualMachine(char *filePath, char *searchStr, char *findStr){
	FILE *fp;
	char str[81];
	fp = fopen(filePath, "r");

	while(!feof(fp))
	{
		fgets(str, 80, fp);
		if(strstr(str, searchStr) != NULL){
			str[strlen(str)-1] = ',';
			strcat(findStr, str);
		}
	}
	fclose(fp);
}

bool IsBlackDevice(char *deviceName)
{
	if(strstr(deviceName, "BlueStacks") != NULL)
	{
		DbgPrint(" BlueStacks : %s,%p", deviceName, deviceName);
		return false;
	}

	if(strstr(deviceName, "Genymotion") != NULL)
	{
		DbgPrint(" Genymotion : %s,%p", deviceName, deviceName);
		return false;
	}
	return true;
}

void CheckVirtualMachine(){
	char findSearch[300] = {0};

	virtualMachine("/system/build.prop", "ro.product.manufacturer", findSearch);
	virtualMachine("/system/build.prop", "ro.product.model", findSearch);
	virtualMachine("/system/build.prop", "ro.product.name", findSearch);
	virtualMachine("/system/build.prop", "ro.product.device", findSearch);

	if(IsBlackDevice(findSearch))	{
		DbgPrint(" BlueStacks : Not Detect");
	}else{
		DbgPrint(" BlueStacks : Detect");
		sleep(10);
		exit(0);
	}
}
