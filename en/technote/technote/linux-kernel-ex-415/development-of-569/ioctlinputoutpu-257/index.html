03.ioctl(Input/Output control)

ioctl(Input/Output control)​

DESCRIPTION​

  • ioctl is an operation provided to control hardware and retrieve status information.
    • While reading and writing data is possible via read() and write(), hardware configuration and status control cannot be achieved with them alone.
    • For example, ioctl is used for operations like:
      • Configuring SPI communication speed or setting an I2C slave address.
      • Instructing a CD-ROM device driver to eject the tray.

SYNOPSIS​

  • The parameter structure of the ioctl function is as follows:
    • The first parameter d (or fd) is the file descriptor obtained from open().
    • The second parameter request is the command passed to the device.
    • Additional arguments can be passed depending on driver implementation.
ioctl()
#include <sys/ioctl.h>
int ioctl(int d, int request, ...);
info
  • https://en.wikipedia.org/wiki/Ioctl
  • http://man7.org/linux/man-pages/man2/ioctl.2.html

Using macros to generate ioctl command numbers​

  • The Linux header /usr/include/asm/ioctl.h defines helper macros used to generate ioctl command numbers.
  • The following command macros are available:
Type
MacroDescription
_IO(int type, int number)Used for simple ioctl calls passing only type and number without data.
_IOR(int type, int number, data_type)Used for ioctl calls that read data from the device driver.
_IOW(int type, int number, data_type)Used for ioctl calls that write data to the device driver.
_IORW(int type, int number, data_type)Used for bidirectional ioctl calls that write and read data.
  • The macro arguments consist of the following:
Argument
ArgumentDescription
type* An 8-bit integer (magic number) unique to the device driver.
* Must be selected so it does not conflict with other drivers that could process the same descriptor.
* For example, TCP and IP stacks use unique numbers because ioctl commands sent to a socket descriptor can be inspected by both layers.
number* An 8-bit integer sequential sequence number.
* Must be uniquely chosen for each command supported within the driver.
data_type* The type name used to calculate the size in bytes exchanged between the user space and driver.
  • Command macros can be defined as follows:
    • SET_DATA is defined with _IOW to write data to the kernel.
    • GET_DATA is defined with _IOR to read data from the kernel.
ioctl_info
struct ioctl_info{
       unsigned long size;
       unsigned int buf[128];
};
 
#define             IOCTL_MAGIC         'G'
#define             SET_DATA			_IOW(IOCTL_MAGIC, 2 , ioctl_info )
#define             GET_DATA			_IOR(IOCTL_MAGIC, 3 , ioctl_info )
  • The following macros can be used to decompose and inspect the fields of a defined ioctl command number:
Macro to read field values
FunctionDescription
_IOC_NR()Macro that extracts the sequence number field
_IOC_TYPE()Macro that extracts the type (magic number) field
_IOC_SIZE()Macro that extracts the data size field
_IOC_DIR()Macro that extracts the read/write direction field

  • http://www.circlemud.org/jelson/software/fusd/docs/node31.html

Source code of module​

chardev_ioctl​

  • When user space opens the device and calls ioctl(), chardev_ioctl() is executed:
    • When cmd is SET_DATA, copy_from_user() copies data passed from user space into the info structure variable.
    • When cmd is GET_DATA, copy_to_user() copies the info structure data stored in the kernel space back to user space.
chardev.c
#include <linux/init.h>
#include <linux/module.h>
#include <linux/types.h>
#include <linux/kernel.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/sched.h>
#include <linux/device.h>
#include <linux/slab.h>
#include <asm/current.h>
#include <linux/uaccess.h>
 
#include "chardev.h"
MODULE_LICENSE("Dual BSD/GPL");
 
#define DRIVER_NAME "chardev"
     
static const unsigned int MINOR_BASE = 0;
static const unsigned int MINOR_NUM  = 1;
static unsigned int chardev_major;
static struct cdev chardev_cdev;
static struct class *chardev_class = NULL;

static int     chardev_open(struct inode *, struct file *);
static int     chardev_release(struct inode *, struct file *);
static ssize_t chardev_read(struct file *, char *, size_t, loff_t *);
static ssize_t chardev_write(struct file *, const char *, size_t, loff_t *);
static long chardev_ioctl(struct file *, unsigned int, unsigned long);

struct file_operations s_chardev_fops = {
    .open    = chardev_open,
    .release = chardev_release,
    .read    = chardev_read,
    .write   = chardev_write,
    .unlocked_ioctl = chardev_ioctl,
};

static int chardev_init(void)
{
    int alloc_ret = 0;
    int cdev_err = 0;
    int minor = 0;
    dev_t dev;
 
    printk("The chardev_init() function has been called.");
     
    alloc_ret = alloc_chrdev_region(&dev, MINOR_BASE, MINOR_NUM, DRIVER_NAME);
    if (alloc_ret != 0) {
        printk(KERN_ERR  "alloc_chrdev_region = %d\n", alloc_ret);
        return -1;
    }
    //Get the major number value in dev.
    chardev_major = MAJOR(dev);
    dev = MKDEV(chardev_major, MINOR_BASE);
 
    //initialize a cdev structure
    cdev_init(&chardev_cdev, &s_chardev_fops);
    chardev_cdev.owner = THIS_MODULE;
 
    //add a char device to the system
    cdev_err = cdev_add(&chardev_cdev, dev, MINOR_NUM);
    if (cdev_err != 0) {
        printk(KERN_ERR  "cdev_add = %d\n", alloc_ret);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }
 
    chardev_class = class_create(THIS_MODULE, "chardev");
    if (IS_ERR(chardev_class)) {
        printk(KERN_ERR  "class_create\n");
        cdev_del(&chardev_cdev);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }
 
    device_create(chardev_class, NULL, MKDEV(chardev_major, minor), NULL, "chardev%d", minor);
    return 0;
}
 
static void chardev_exit(void)
{
    int minor = 0; 
    dev_t dev = MKDEV(chardev_major, MINOR_BASE);
     
    printk("The chardev_exit() function has been called.");

    device_destroy(chardev_class, MKDEV(chardev_major, minor));
 
    class_destroy(chardev_class);
    cdev_del(&chardev_cdev);
    unregister_chrdev_region(dev, MINOR_NUM);
}

static int chardev_open(struct inode *inode, struct file *file)
{
    printk("The chardev_open() function has been called.");
    return 0;
}
 
static int chardev_release(struct inode *inode, struct file *file)
{
    printk("The chardev_close() function has been called.");
    return 0;
}
 
static ssize_t chardev_write(struct file *filp, const char __user *buf, size_t count, loff_t *f_pos)
{
    printk("The chardev_write() function has been called.");   
    return count;
}
 
static ssize_t chardev_read(struct file *filp, char __user *buf, size_t count, loff_t *f_pos)
{
    printk("The chardev_read() function has been called.");
    return count;
}
 
static struct ioctl_info info;
static long chardev_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    printk("The chardev_ioctl() function has been called.");
 
    switch (cmd) {
        case SET_DATA:
            printk("SET_DATA\n");
            if (copy_from_user(&info, (void __user *)arg, sizeof(info))) {
                return -EFAULT;
            }
	    printk("info.size : %ld, info.buf : %s",info.size, info.buf);
            break;
        case GET_DATA:
            printk("GET_DATA\n");
            if (copy_to_user((void __user *)arg, &info, sizeof(info))) {
                return -EFAULT;
            }
            break;
        default:
            printk(KERN_WARNING "unsupported command %d\n", cmd);
 
        return -EFAULT;
    }
    return 0;
}

module_init(chardev_init);
module_exit(chardev_exit);
  • The macro definition is as follows:
    • SET_DATA is set to _IOW (write) taking struct ioctl_info as the argument type.
    • GET_DATA is set to _IOR (read) taking struct ioctl_info as the argument type.
chardev.h
#ifndef CHAR_DEV_H_
#define CHAR_DEV_H_
#include <linux/ioctl.h>
 
struct ioctl_info{
       unsigned long size;
       char buf[128];
};
   
#define             IOCTL_MAGIC         'G'
#define             SET_DATA            _IOW(IOCTL_MAGIC, 2 ,struct ioctl_info)
#define             GET_DATA            _IOR(IOCTL_MAGIC, 3 ,struct ioctl_info)
 
#endif

Source code of Test program​

  • The test program operates as follows:
    • Uses open() to open /dev/chardev0 and obtain a file descriptor fd.
    • Uses ioctl() with SET_DATA and passes &set_info to copy user space data into the kernel.
    • Uses ioctl() with GET_DATA and passes &get_info to copy kernel data into user space.
    • Uses printf() to print the retrieved data.
    • Uses close() to close fd.
test.c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <sys/ioctl.h> 
#include "chardev.h"
 
int main()
{
    int fd;
    struct ioctl_info set_info; 
    struct ioctl_info get_info; 
 
 	set_info.size = 100;
	strncpy(set_info.buf,"lazenca.0x0",11);

    if ((fd = open("/dev/chardev0", O_RDWR)) < 0){
        printf("Cannot open /dev/chardev0. Try again later.\n");
    }
 
    if (ioctl(fd, SET_DATA, &set_info) < 0){
        printf("Error : SET_DATA.\n");
    }

    if (ioctl(fd, GET_DATA, &get_info) < 0){
        printf("Error : SET_DATA.\n");
    }
 
 	printf("get_info.size : %ld, get_info.buf : %s\n", get_info.size, get_info.buf);
 
    if (close(fd) != 0){
        printf("Cannot close.\n");
    }
    return 0;
}
